# ---------------------------------------------------------------------------
# Hybrid-attention kernels for Qwen3.8-27B. Installed by
# scripts/setup_qwen38_kernels.sh because both build from source.
#
# Qwen3.8-27B is hybrid: 48 linear_attention + 16 full_attention layers.
# transformers/models/qwen3_5/modeling_qwen3_5.py gates these behind
# is_causal_conv1d_available() / is_flash_linear_attention_available() and
# SILENTLY falls back to slow pure-torch ops when either is missing.
# flash-linear-attention is held at 0.4.2: 0.5.0 produces "!!!!" gibberish via
# the gated-delta-rule kernels (fla-org/flash-linear-attention#792).
causal-conv1d==1.6.2.post1
flash-linear-attention==0.4.2

# NOTE on flash-attn: intentionally NOT pinned here, and not required.
#   - vLLM rollouts use vLLM's OWN bundled FA2 kernels (logs show
#     "Using FLASH_ATTN attention backend" / "FlashAttention version 2"),
#     independent of this package.
#   - transformers training uses ATTN_IMPLEMENTATION=sdpa, and torch SDPA
#     dispatches to its own FlashAttention-2 backend at head_dim=256 on Hopper.
#   - Building it here FAILS: host CUDA is 13.2 vs torch cu128.
# See scripts/setup_flash_attn.sh for the full diagnosis and the only route
# that would work (a real CUDA 12.8 toolkit).
